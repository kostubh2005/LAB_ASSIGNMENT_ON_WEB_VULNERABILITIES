// HTML-escape helper (context-aware output encoding)
function escapeHtml(str) {
  return String(str)
    .replace(/&/g,  "&amp;")
    .replace(/</g,  "&lt;")
    .replace(/>/g,  "&gt;")
    .replace(/"/g,   "&quot;")
    .replace(/'/g,   "&#39;");
}

// Route: GET /safe?q=…
app.get("/safe", (req, res) => {
